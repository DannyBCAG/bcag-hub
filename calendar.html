const SPREADSHEET_ID = '14bMuA-ilr-yzTNUreRbKQi01uU-yb3q67jFD6jPLqQk'; function doGet() { const spreadsheet = SpreadsheetApp.openById(SPREADSHEET_ID); // CALENDAR const calendarSheet = spreadsheet.getSheetByName('Calendar'); const calendarValues = calendarSheet.getDataRange().getDisplayValues(); calendarValues.shift(); const calendar = calendarValues .filter(row => row.some(cell => cell.trim() !== '')) .map(row => ({ date: row[0], time: row[1], status: row[2], notes: row[3] })); // MATERIALS const materialsSheet = spreadsheet.getSheetByName('Materials'); const materialsValues = materialsSheet.getDataRange().getDisplayValues(); materialsValues.shift(); const materials = materialsValues .map(row => row[0]) .filter(item => item.trim() !== ''); // SEND BOTH return ContentService .createTextOutput(JSON.stringify({ calendar: calendar, materials: materials })) .setMimeType(ContentService.MimeType.JSON); }
